\begin{frame}[plain]
  \vspace{0.15cm}
  {\usebeamerfont{title}\color{RlmNavy}
    Can a model build an unfamiliar solution from familiar pieces?\par}

  \vspace{0.15cm}
  {\usebeamerfont{subtitle}\color{RlmMuted}
    Preliminary and exploratory research update\par}

  \vspace{0.4cm}
  \begin{beamercolorbox}[wd=\linewidth,sep=.7em,rounded=true]{block body}
    \centering
    \large
    Language models are strongest when a problem resembles something they have seen before.\\[0.45em]
    We are studying whether a Recursive Language Model (RLM) can turn a new, difficult problem
    into \textbf{smaller, more familiar problems}.
  \end{beamercolorbox}

  \vfill
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \centering
      {\color{RlmTeal}\textbf{We completed several preliminary training studies.}}\\[0.25em]
      The model learned several ways to work inside the RLM.
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      {\color{RlmGold}\textbf{The larger question remains open.}}\\[0.25em]
      Can this process help on genuinely new problems?
    \end{column}
  \end{columns}
\end{frame}

% Slide 2 -------------------------------------------------------------------
\begin{frame}{An RLM gives the model a workspace and a way to hand off work.}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.205\textwidth}
      \rlmstagebox{\textbf{The full problem}\\[.2em]
        \footnotesize It stays in a Python workspace.}
    \end{column}
    \begin{column}{.04\textwidth}
      \centering\Large\color{RlmTeal}$\rightarrow$
    \end{column}
    \begin{column}{.23\textwidth}
      \rlmfocusbox{\textbf{The main model}\\[.2em]
        \footnotesize It inspects selected parts and chooses what to do.}
    \end{column}
    \begin{column}{.04\textwidth}
      \centering\Large\color{RlmTeal}$\rightarrow$
    \end{column}
    \begin{column}{.23\textwidth}
      \rlmstagebox{\textbf{Focused model calls}\\[.2em]
        \footnotesize They work on selected pieces.}
    \end{column}
    \begin{column}{.04\textwidth}
      \centering\Large\color{RlmTeal}$\rightarrow$
    \end{column}
    \begin{column}{.185\textwidth}
      \rlmstagebox{\textbf{One answer}\\[.2em]
        \footnotesize It combines the results.}
    \end{column}
  \end{columns}

  \vspace{.25cm}
  \centering
  {\small\color{RlmGold}\textbf{A focused call can repeat the same process.}}

  \vspace{0.35cm}
  \begin{block}{The main model may never need to read the entire original input at once.}
    It can use code to inspect selected parts, ask other model calls to work on those parts, and
    combine the returned results. The Python workspace holds the large input and the intermediate
    work.
  \end{block}
\end{frame}

% Slide 3 -------------------------------------------------------------------
\begin{frame}{The larger idea: solve a new whole by combining familiar pieces.}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.22\textwidth}
      \rlmstagebox{\textbf{An unfamiliar problem}\\[.25em]
        \small The whole problem does not resemble one training example.}
    \end{column}
    \begin{column}{.05\textwidth}
      \centering\Large\color{RlmTeal}$\rightarrow$
    \end{column}
    \begin{column}{.42\textwidth}
      \rlmfocusbox{\textbf{The model chooses how to divide it}\\[.35em]
        \small Familiar piece 1\\[.15em]
        Familiar piece 2\\[.15em]
        Familiar piece 3}
    \end{column}
    \begin{column}{.05\textwidth}
      \centering\Large\color{RlmTeal}$\rightarrow$
    \end{column}
    \begin{column}{.22\textwidth}
      \rlmstagebox{\textbf{A combined answer}\\[.25em]
        \small The RLM brings the smaller answers together.}
    \end{column}
  \end{columns}

  \vspace{0.35cm}
  \begin{block}{Researchers call this compositional generalization.}
    Each call may only need to solve a local problem that resembles its training experience. The
    RLM keeps track of the work, routes the pieces, and combines the answers. We ask when this
    division works and whether training can improve it.
  \end{block}

  \vfill
  {\tiny\color{RlmMuted}
    Sources: \href{https://alexzhang13.github.io/blog/2025/rlm/}{Zhang (2025), ``Recursive
    Language Models''}; \href{https://alexzhang13.github.io/blog/2026/harness/}{Zhang (2026),
    ``Harnesses are compositional generalizers.''}}
\end{frame}

% Slide 4 -------------------------------------------------------------------
\begin{frame}{We can now repeat the full supervised train-and-test process.}
  \begin{beamercolorbox}[wd=\linewidth,sep=.75em,rounded=true]{block body}
    \textbf{Supervised fine-tuning (SFT)} means showing the model good worked examples and
    training it to imitate the useful decisions in those examples.
  \end{beamercolorbox}

  \vspace{0.5cm}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.21\textwidth}
      \rlmstagebox{\textbf{1. Create}\\[.2em]
        \small Check examples of good RLM work.}
    \end{column}
    \begin{column}{.04\textwidth}
      \centering\Large\color{RlmTeal}$\rightarrow$
    \end{column}
    \begin{column}{.21\textwidth}
      \rlmfocusbox{\textbf{2. Train}\\[.2em]
        \small Train the model on those examples.}
    \end{column}
    \begin{column}{.04\textwidth}
      \centering\Large\color{RlmTeal}$\rightarrow$
    \end{column}
    \begin{column}{.21\textwidth}
      \rlmstagebox{\textbf{3. Load}\\[.2em]
        \small Put a saved version inside the RLM.}
    \end{column}
    \begin{column}{.04\textwidth}
      \centering\Large\color{RlmTeal}$\rightarrow$
    \end{column}
    \begin{column}{.21\textwidth}
      \rlmstagebox{\textbf{4. Test}\\[.2em]
        \small Check its answers and its steps.}
    \end{column}
  \end{columns}

  \vspace{0.4cm}
  \begin{block}{This is now a repeatable experiment, not just a demonstration.}
    We can save and test exact model versions, compare training methods, and measure whether
    improvements carry over to genuinely new tasks.
  \end{block}
\end{frame}

\begin{frame}{What did SFT teach? A simple two-step RLM routine.}
  \rlmstagebox{\small\textbf{Starting point:} The task is in the Python workspace, outside the

  \vspace{.12cm}
  \begin{columns}[c,onlytextwidth]
    \begin{column}{.28\textwidth}
      \rlmfocusbox{\textbf{1. Model action}\\[.12em]
        \footnotesize The desired model output inspects the workspace with Python.}
    \end{column}
    \begin{column}{.04\textwidth}
      \centering\Large\color{RlmTeal}$\rightarrow$
    \end{column}
    \begin{column}{.32\textwidth}
      \rlmstagebox{\textbf{2. RLM observation}\\[.12em]
        \footnotesize ``Count the numbers divisible by 5 in
        \texttt{[5, 8, 10, 21, 25]}.''}
    \end{column}
    \begin{column}{.04\textwidth}

    \end{column}
    \begin{column}{.28\textwidth}

% Slide 6 -------------------------------------------------------------------
\begin{frame}{A harder test showed what transferred---and what did not.}
  \rlmstagebox{\textbf{Simplified local example}\\[.2em]
    \small Records \texttt{A:4, B:3, A:2}. Add the values within each group:
    \texttt{A:6, B:3}.}

  \vspace{.25cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \centering
      \rlmfocusbox{\textbf{Local work transferred}\\[.35em]
        \small Inspect the data\\[-.05em]
        {\color{RlmTeal}$\downarrow$}\\[-.05em]
        Calculate, remember, and check\\[.4em]
        {\Large\color{RlmNavy}\textbf{24/24 tasks}}}
    \end{column}

    \begin{column}{0.48\textwidth}
      \centering
      \rlmnextbox{\textbf{Model-to-model handoffs did not}\\[.35em]
        \small Send pieces to focused model calls\\[-.05em]
        {\color{RlmGold}$\downarrow$}\\[-.05em]
        Combine their returned answers\\[.4em]
        {\Large\color{RlmRed}\textbf{0/12 tasks}}}
    \end{column}
  \end{columns}

  \vspace{0.25cm}
  \begin{beamercolorbox}[wd=\linewidth,sep=.6em,rounded=true]{block body}
    \centering
    \textbf{Small exploratory test: six examples per workflow.} Local RLM work transferred, but
    recursive handoffs did not.
  \end{beamercolorbox}
\end{frame}

  \vspace{0.35cm}
  \begin{block}{The comparison is designed to answer a simple question.}
    Does practice inside the RLM teach something beyond ordinary task practice? We will compare a
    model trained on direct solutions with a model trained on successful RLM work.
  \end{block}

\end{frame}

% Slide 8 -------------------------------------------------------------------
\begin{frame}{The next experiments will test the larger idea directly.}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \begin{block}{Does learning carry over?}
        Does learning carry to new examples, formats, and eventually new kinds of problem?
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{What caused improvement?}
        We will compare dividing the problem with ordinary practice and with simply giving the
        model more calls.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Can scores teach the model?}
        If scores separate good and bad attempts, reward-based training can improve the model's
        decisions.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.3cm}
  \begin{beamercolorbox}[wd=\linewidth,sep=.6em,rounded=true]{block body}
    \centering
    \large
    \textbf{The big picture}\par
    \vspace{0.35em}
    We are testing whether training a model to use the RLM can help it solve unfamiliar
    problems by building solutions from familiar pieces.
  \end{beamercolorbox}
